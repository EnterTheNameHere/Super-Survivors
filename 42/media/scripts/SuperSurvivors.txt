        DisplayName = Grave,

        Tooltip = RIP,
        ResizeWorldIcon = 0.4,
    }

    item GraveCross
    {
        Type = Normal,
        DisplayName = Grave,
        Icon = gravecross,
        Weight = 50,
        Tooltip = RIP,
        ResizeWorldIcon = 0.4,
    }

    timedAction SSRDigGrave

    craftRecipe DigGrave
    {
        timedAction = Survivors.SSRDigGrave,

            item 1 [Base.CorpseMale;Base.CorpseFemale],
            item 1 [Base.Shovel] mode:keep flags[Prop1],

        {
            item 1 Survivors.Grave,

    }

    craftRecipe AddCrossToGrave
    {
        timedAction = Survivors.SSRAddGraveCross,

        {
            item 1 [Survivors.Grave],
            item 2 [Base.Plank] flags[Prop2],
            item 1 [Base.Nails],
            item 1 [Base.Hammer] mode:keep flags[Prop1],
        }
        outputs
        {
            item 1 Survivors.GraveCross,

    }

    craftRecipe DigGraveWithCross
    {
        timedAction = Survivors.SSRDigGrave,
